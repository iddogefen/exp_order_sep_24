# 120 - 2026-10-05 # Rank ALL 48 pairs by P(correct causal direction) among subjects who passed the direct-test screen. # Pairs 26 and 40 had their cause/effect images SWAPPED in the pilot (since fixed), so for those two pairs # the judgments are REVERSED: a pilot "effect_to_cause" answer = the intended cause -> effect direction. import os, glob import pandas as pd DATA_DIR = os.environ.get("DATA_DIR", "/content/drive/MyDrive/phd_data_2026/data_full_2026/Value_Data_2026_7/ValueSpreading_WithinSubject_CauseEffect_EffectCause_v1_2026-10-05_data") DIRECT_THRESHOLD = 0.70 SWAPPED_IN_PILOT = [26, 40] # judgments flipped for these pairs REMOVED_NOW = [8, 18, 26, 32, 33, 40, 42, 48] # current list in experiment_40_pairs.html N_KEEP = 40 files = sorted(glob.glob(os.path.join(DATA_DIR, "*exp_data*.csv"))) frames = [] for f in files: d = pd.read_csv(f, low_memory=False) if "subject_id" in d.columns and d["subject_id"].notna().any(): d["subject_id"] = d["subject_id"].dropna().iloc[0] frames.append(d) raw = pd.concat(frames, ignore_index=True) raw = raw.drop_duplicates(subset=["subject_id", "trial_index"]) # direct-test screen: same definition as before (trials linked to 26/40 left out, so the passers stay the same) dec = raw[raw["test_type"] == "direct"].copy() for c in ("pair_id", "reward_pair_id", "neutral_pair_id"): dec = dec[~pd.to_numeric(dec[c], errors="coerce").isin(SWAPPED_IN_PILOT)] dec["chose_reward"] = pd.to_numeric(dec["chose_reward"], errors="coerce") acc = dec.groupby("subject_id")["chose_reward"].mean() passers = acc[acc > DIRECT_THRESHOLD].index print(f"{len(passers)} of {len(acc)} subjects pass the screen (> {DIRECT_THRESHOLD}).") cj = raw[(raw["task"] == "causal_judgment") & raw["subject_id"].isin(passers)].copy() cj["pair_id"] = pd.to_numeric(cj["pair_id"], errors="coerce").astype(int) cj["judgment"] = cj["judgment"].astype(str).str.strip() # reverse the direction for the swapped pairs flip = {"cause_to_effect": "effect_to_cause", "effect_to_cause": "cause_to_effect"} sw = cj["pair_id"].isin(SWAPPED_IN_PILOT) cj.loc[sw, "judgment"] = cj.loc[sw, "judgment"].map(lambda j: flip.get(j, j)) cj["c2e"] = (cj["judgment"] == "cause_to_effect").astype(int) t = cj.groupby("pair_id")["c2e"].agg(["sum", "count"]).rename(columns={"sum": "cause_to_effect", "count": "n"}) t["p"] = t["cause_to_effect"] / t["n"] t = t.sort_values(["p", "n"], ascending=[False, False]) t["rank"] = range(1, len(t) + 1) t["status_now"] = ["REMOVED" if p in REMOVED_NOW else "in the 40" for p in t.index] t["note"] = ["judgments reversed (swapped in pilot)" if p in SWAPPED_IN_PILOT else "" for p in t.index] print(t.to_string()) print("\n26 and 40 after reversing:") print(t.loc[[p for p in SWAPPED_IN_PILOT if p in t.index]].to_string()) best = list(t.index[:N_KEEP]) cur = [p for p in t.index if p not in REMOVED_NOW] print("\nBest 40 by data:", sorted(best)) print("Current 40 :", sorted(cur)) print("In current but not in best-40 :", sorted(set(cur) - set(best))) print("In best-40 but currently removed:", sorted(set(best) - set(cur))) print("Pairs NOT in best-40 (would be removed):", sorted(set(t.index) - set(best))) print("\nNote: the cut-off falls inside a tie; with this n, differences of 1 judgment are not meaningful.")
